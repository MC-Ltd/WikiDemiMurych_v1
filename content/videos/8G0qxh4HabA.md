'use strict';


function doMain (){

    ============ на одном уровне =============================

    var theVar = 1;
    let theLet = 2;
    const theConst = 3;
    let sumator = 0;

    for (var i = 100000000; i > 0; i--){
        sumator += (theVar + theLet + theConst)
    }
    return sumator;

    ============= на разных уровнях ===========================

    function doInsideMain (){
        for (var i = 100000000; i > 0; i--){
            sumator += (theVar + theLet + theConst)
        }
        return sumator;
     }

    doInsideMain();

    ============ на разных уровнях, но с параметрами ============

    function doInsideMain (theVar, theLet, theConst, sumator){
        var a = theVar;
        let b = theLet;
        const c = theConst;
        let summ = sumator;
        
        for (var i = 100000000; i > 0; i--){
            summ += (a + b + c)
        }
        return summ;
    }

    doInsideMain();

}
