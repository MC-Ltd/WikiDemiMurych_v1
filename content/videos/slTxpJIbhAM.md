const sumOfOnlyUnique = array => {
    var map = new Map();
    return array.reduce(
        (acc, cur) => {
            const count = map.get(cur) ?? 0;
            map.set(cur, count + 1);
            if (count === 0) acc += cur;
            if (count === 1) acc -= cur;
            return acc;
        },
        0
    );
};
